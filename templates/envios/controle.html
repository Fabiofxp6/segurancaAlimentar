function iniciarEnvios(painel, confirmar, acompanhar = () => {}) { const aviso = painel.querySelector("[data-estado-conexao]"); const painelConexao = painel.querySelector("[data-painel-conexao]"); const verificarConexao = painel.querySelector("[data-verificar-conexao]"); const titulo = painel.dataset.tituloSemConexao ? document.querySelector(".barra-app__titulo") : null; const tituloNormal = titulo?.textContent; const controles = painel.querySelectorAll("form[data-envio]"); const componentes = new Map([...controles].map((form) => [form, { campos: form.querySelector("[data-campos]"), acao: form.querySelector('button[type="submit"]'), rotulo: form.querySelector("[data-status]"), }])); const acoesExternas = document.querySelectorAll("[data-acao-servidor]"); const cabecalhos = {"X-Contexto-Campo": painel.dataset.contexto}; let pendente = null; let semConexao = !navigator.onLine; let encerrado = false; let enviando = false; let sondando = false; let timer; let intervalo = 1500; let navegacaoBloqueada; function estado(form, situacao, mensagem) { const {rotulo} = componentes.get(form); if (rotulo.dataset.estado !== situacao) rotulo.dataset.estado = situacao; if (rotulo.textContent !== mensagem) rotulo.textContent = mensagem; } function avisoConexao(mensagem) { aviso.textContent = mensagem; aviso.hidden = !mensagem; if (painelConexao) painelConexao.hidden = !mensagem; if (verificarConexao) verificarConexao.hidden = encerrado; if (titulo) titulo.textContent = semConexao ? painel.dataset.tituloSemConexao : tituloNormal; } function chaveDeEnvio() { if (crypto.randomUUID) return crypto.randomUUID(); const bytes = crypto.getRandomValues(new Uint8Array(16)); bytes[6] = (bytes[6] & 0x0f) | 0x40; bytes[8] = (bytes[8] & 0x3f) | 0x80; const hexadecimal = Array.from(bytes, (byte) => byte.toString(16).padStart(2, "0")).join(""); return [hexadecimal.slice(0, 8), hexadecimal.slice(8, 12), hexadecimal.slice(12, 16), hexadecimal.slice(16, 20), hexadecimal.slice(20)].join("-"); } function atualizarControles() { const bloquearNavegacao = semConexao || Boolean(pendente); for (const [form, {campos, acao}] of componentes) { const bloqueado = form.dataset.liberado === "nao"; const envioAtual = pendente?.form === form; campos.disabled = bloqueado || envioAtual || encerrado; acao.disabled = bloqueado || bloquearNavegacao || encerrado; const ocupado = envioAtual && enviando && !semConexao; campos.setAttribute("aria-busy", String(ocupado)); } if (navegacaoBloqueada === bloquearNavegacao) return; navegacaoBloqueada = bloquearNavegacao; for (const botao of acoesExternas) botao.disabled = bloquearNavegacao; for (const link of document.querySelectorAll("a[href]:not([data-nav-local])")) { if (bloquearNavegacao) link.setAttribute("aria-disabled", "true"); else link.removeAttribute("aria-disabled"); } } function agendar() { clearTimeout(timer); if (semConexao && !encerrado) { timer = setTimeout(recuperar, intervalo); intervalo = Math.min(intervalo * 2, 10000); } } function perderConexao() { semConexao = true; avisoConexao("Sem conexão com o servidor. O preenchimento permanece nesta tela. Dados pendentes ainda não foram salvos."); if (pendente) { estado(pendente.form, "pendente", "Pendente · aguardando reconexão"); acompanhar(pendente.form, "pendente"); } atualizarControles(); agendar(); } function encerrarEnvio() { encerrado = true; semConexao = false; clearTimeout(timer); if (pendente) { estado(pendente.form, "falha", "Falha · sessão ou acesso mudou"); acompanhar(pendente.form, "falha"); } pendente = null; avisoConexao("A sessão ou o acesso mudou. Confira o preenchimento antes de reabrir a página."); atualizarControles(); } async function requisicao(url, opcoes) { const controle = new AbortController(); const limite = setTimeout(() => controle.abort(), 12000); try { const http = await fetch(url, { cache: "no-store", credentials: "same-origin", redirect: "error", headers: cabecalhos, ...opcoes, signal: controle.signal, }); const json = http.headers.get("Content-Type")?.includes("application/json") ? await http.json() : null; return {http, json}; } finally { clearTimeout(limite); } } async function enviar() { if (!pendente || semConexao || enviando || encerrado) return; const envio = pendente; enviando = true; atualizarControles(); estado(envio.form, "enviando", "Enviando"); acompanhar(envio.form, "enviando"); try { const {http, json} = await requisicao(envio.form.action, {method: "POST", body: envio.dados}); if ([401, 403, 404].includes(http.status)) { encerrarEnvio(); } else if ([400, 409, 413, 422].includes(http.status)) { estado(envio.form, "falha", "Falha · " + (json?.erro || "Confira os dados antes de tentar novamente.")); acompanhar(envio.form, "falha"); pendente = null; } else if (http.ok && json?.operacao_id === envio.chave && json.confirmado_em) { confirmar(envio.form, json); acompanhar(envio.form, null); estado(envio.form, "salva", "Salva"); envio.form.dispatchEvent(new Event("envio-confirmado")); pendente = null; } else { perderConexao(); } } catch { perderConexao(); } finally { enviando = false; atualizarControles(); } } async function recuperar() { if (!semConexao || encerrado || sondando) return; if (!navigator.onLine || enviando) { agendar(); return; } sondando = true; if (verificarConexao) verificarConexao.disabled = true; try { const {http, json} = await requisicao(painel.dataset.conexao); if ([401, 403, 404].includes(http.status)) { encerrarEnvio(); } else if (http.ok && json?.conectado === true) { semConexao = false; intervalo = 1500; clearTimeout(timer); avisoConexao(""); atualizarControles(); void enviar(); } else { agendar(); } } catch { agendar(); } finally { sondando = false; if (verificarConexao) verificarConexao.disabled = false; } } for (const form of controles) { form.addEventListener("input", () => estado(form, "pendente", "Pendente")); form.addEventListener("submit", (evento) => { evento.preventDefault(); if (form.dataset.liberado === "nao" || pendente || semConexao || encerrado) return; const dados = new FormData(form); const chave = chaveDeEnvio(); dados.set("operacao_id", chave); pendente = {form, dados, chave}; atualizarControles(); void enviar(); }); } document.addEventListener("click", (evento) => { if (evento.target.closest('a[aria-disabled="true"]')) evento.preventDefault(); }); verificarConexao?.addEventListener("click", recuperar); addEventListener("offline", perderConexao); addEventListener("online", recuperar); atualizarControles(); if (semConexao) perderConexao(); return atualizarControles; }
