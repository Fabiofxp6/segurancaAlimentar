function chavePendencias(painel) { return "inspecao-envios:" + painel.dataset.contexto + ":" + painel.dataset.inspecao; } function lerPendencias(painel) { try { const dados = JSON.parse(sessionStorage.getItem(chavePendencias(painel)) || "{}"); if (Array.isArray(dados)) return new Map(dados.map((chave) => [chave, "pendente"])); return new Map(Object.entries(dados)); } catch { return new Map(); } } function registrarPendencia(painel, tipo, situacao) { try { const chave = painel.dataset.item + ":" + tipo; const pendencias = lerPendencias(painel); if (situacao) pendencias.set(chave, situacao); else pendencias.delete(chave); if (pendencias.size) { sessionStorage.setItem(chavePendencias(painel), JSON.stringify(Object.fromEntries(pendencias))); } else { sessionStorage.removeItem(chavePendencias(painel)); } } catch { // O envio ainda exige a confirmação do servidor. } }
